% --- Furigana macro (no extra packages needed, works with plain pdflatex+CJK) ---
% Usage: \rb{漢字}{かんじ} stacks the reading in small kana above the kanji.
% Built as a two-row \vbox rather than a \raisebox: CJK glyphs are full-height,
% so a fixed raise drops the kana on top of the kanji. The vbox also reports its
% true height to TeX, so the line below gets pushed down instead of collided
% into. Centres whichever row is narrower over the wider one.
\newcommand{\rb}[2]{%
  \leavevmode
  \begingroup
  \setbox0=\hbox{#1}%
  \setbox2=\hbox{\tiny #2}%
  \dimen0=\wd0 \ifdim\wd2>\dimen0 \dimen0=\wd2 \fi
  \vbox{\offinterlineskip
    \hbox to\dimen0{\hfil\tiny #2\hfil}%
    \kern0.2ex
    \hbox to\dimen0{\hfil\unhbox0\hfil}%
  }%
  \endgroup
}
